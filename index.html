  function T(key, vars) {

    var pack = I18N[state.lang] || I18N.zh;

    var s = pack[key];

    if (s == null) s = I18N.zh[key];

    if (s == null) return key;

    if (vars) {

      s = s.replace(/\{(\w+)\}/g, function (m, name) {

        return vars[name] == null ? m : String(vars[name]);

      });

    }

    return s;

  }

  function load() {

    var d = defaults();

    try {

      var raw = localStorage.getItem(STORE_KEY);

      if (!raw) return d;

      var s = JSON.parse(raw);

      if (!s || typeof s !== 'object') return d;

      if (s.lang === 'en' || s.lang === 'zh') d.lang = s.lang;

      if (Array.isArray(s.plans)) d.plans = s.plans.filter(function (p) { return p && typeof p === 'object'; });

      if (Array.isArray(s.bills)) d.bills = s.bills.filter(function (b) { return b && typeof b === 'object'; });

      if (typeof s.members === 'number' && s.members >= 1) d.members = Math.min(20, Math.floor(s.members));

      if (s.repair && typeof s.repair === 'object') {

        Object.keys(d.repair).forEach(function (k) {

          if (typeof s.repair[k] === typeof d.repair[k]) d.repair[k] = s.repair[k];

        });

        d.repair.step = Math.min(4, Math.max(1, parseInt(d.repair.step, 10) || 1));

        if (!d.repair.scenario) d.repair.step = 1;

      }

    } catch (e) { /* 数据损坏时使用默认值 */ }

    return d;

  }

  function dateLabel(str) {

  function fallbackCopy(text) {

    var ta = document.createElement('textarea');

    ta.value = text;

    ta.setAttribute('readonly', '');

    ta.style.position = 'fixed';

    ta.style.top = '-1000px';

    ta.style.opacity = '0';

    document.body.appendChild(ta);

    ta.select();

    ta.setSelectionRange(0, ta.value.length);

    var ok = false;

    try { ok = document.execCommand('copy'); } catch (e) { ok = false; }

    document.body.removeChild(ta);

    return ok;

  }


  function copyText(text) {

    if (navigator.clipboard && window.isSecureContext) {

      return navigator.clipboard.writeText(text).then(function () { return true; })

        .catch(function () { return fallbackCopy(text); });

    }

    return Promise.resolve(fallbackCopy(text));

  }

  /* ---- 拖拽 ---- */


  var drag = null;

  var hlCell = null;


  document.addEventListener('pointerdown', function (e) {

    if (e.button && e.button !== 0) return;

    var card = e.target.closest ? e.target.closest('.plan') : null;

    if (!card) return;

    drag = { id: card.dataset.id, card: card, x: e.clientX, y: e.clientY, moved: false, ox: 0, oy: 0, ghost: null };

  });


  document.addEventListener('pointermove', function (e) {

    if (!drag) return;

    if (!drag.moved) {

      if (Math.abs(e.clientX - drag.x) < 8 && Math.abs(e.clientY - drag.y) < 8) return;

      drag.moved = true;

      var rect = drag.card.getBoundingClientRect();

      var ghost = drag.card.cloneNode(true);

      ghost.classList.add('ghost');

      ghost.classList.remove('dragging');

      ghost.style.width = rect.width + 'px';

      ghost.style.height = rect.height + 'px';

      document.body.appendChild(ghost);

      drag.ghost = ghost;

      drag.ox = e.clientX - rect.left;

      drag.oy = e.clientY - rect.top;

      drag.card.classList.add('dragging');

      document.body.classList.add('dragging-now');

    }

    e.preventDefault();

    drag.ghost.style.transform = 'translate(' + (e.clientX - drag.ox) + 'px,' + (e.clientY - drag.oy) + 'px)';


    var el = document.elementFromPoint(e.clientX, e.clientY);

    var cell = el && el.closest ? el.closest('.board-cell') : null;

    if (cell !== hlCell) {

      if (hlCell) hlCell.classList.remove('hl');

      if (cell) cell.classList.add('hl');

      hlCell = cell;

    }


    var sc = $('.board-scroll');

    if (sc) {

      var sr = sc.getBoundingClientRect();

      if (e.clientX < sr.left + 40) sc.scrollLeft -= 16;

      else if (e.clientX > sr.right - 40) sc.scrollLeft += 16;

    }

  }, { passive: false });


  function endDrag(e) {

    if (!drag) return;

    var d = drag;

    drag = null;

    if (hlCell) { hlCell.classList.remove('hl'); hlCell = null; }

    document.body.classList.remove('dragging-now');

    if (!d.moved) return;


    d.card.classList.remove('dragging');

    if (d.ghost && d.ghost.parentNode) d.ghost.parentNode.removeChild(d.ghost);

    ui.suppressClickUntil = Date.now() + 350;


    var el = document.elementFromPoint(e.clientX, e.clientY);

    var cell = el && el.closest ? el.closest('.board-cell') : null;

    var p = state.plans.filter(function (x) { return x.id === d.id; })[0];


    if (cell && p) {

      var newDate = cell.dataset.date;

      var newSlot = parseInt(cell.dataset.slot, 10) || 0;

      if (p.date !== newDate || p.slot !== newSlot) {

        p.date = newDate;

        p.slot = newSlot;

        save();

        renderFacility();

        toast(T('toastMoved', { when: dateLabel(p.date), slot: SLOTS[p.slot] }));

        return;

      }

    }

    renderFacility();

  }

        + '<span class="lname">' + esc(b.name) + '</span>'

        + '<span class="lpct">' + pct.toFixed(0) + '%</span></div>';

    }).join('');

  }


  function drawPie() {

    var canvas = document.getElementById('pieCanvas');

    if (!canvas) return;

    var wrap = canvas.parentNode;

    var w = Math.max(180, wrap.clientWidth || 260);

    var h = 220;

    var dpr = window.devicePixelRatio || 1;

    canvas.width = Math.round(w * dpr);

    canvas.height = Math.round(h * dpr);

    canvas.style.width = w + 'px';

    canvas.style.height = h + 'px';


    var ctx = canvas.getContext('2d');

    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    ctx.clearRect(0, 0, w, h);


    var cx = w / 2, cy = h / 2, r = Math.min(w, h) / 2 - 12;

    var items = state.bills.filter(function (b) { return (Number(b.amount) || 0) > 0; });

    var total = items.reduce(function (s, b) { return s + Number(b.amount); }, 0);


    if (!items.length || total <= 0) {

      ctx.beginPath();

      ctx.arc(cx, cy, r, 0, Math.PI * 2);

      ctx.fillStyle = '#eef1f5';

      ctx.fill();

      ctx.fillStyle = '#9aa6b5';

      ctx.font = '13px -apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';

      ctx.textAlign = 'center';

      ctx.textBaseline = 'middle';

      ctx.fillText(T('canvasEmpty'), cx, cy);

      return;

    }


    var start = -Math.PI / 2;

    items.forEach(function (b) {

      var idx = state.bills.indexOf(b);

      var ang = Number(b.amount) / total * Math.PI * 2;

      ctx.beginPath();

      ctx.moveTo(cx, cy);

      ctx.arc(cx, cy, r, start, start + ang);

      ctx.closePath();

      ctx.fillStyle = BILL_COLORS[idx % BILL_COLORS.length];

      ctx.fill();

      ctx.lineWidth = 2;

      ctx.strokeStyle = '#fff';

      ctx.stroke();

      start += ang;

    });

  }

  function buildBillsText() {

    var m = state.members;

    var total = billsTotal();

    var lines = [T('billsTextHead', { m: m }), ''];

    if (state.bills.length) {

        lines.push((i + 1) + '. ' + b.name + (isEn() ? ': ' : '：') + fmtMoney(b.amount) + T('billsTextEach', { v: fmtMoney((Number(b.amount) || 0) / m) }));

    }

    return lines.join('\n');

  }
